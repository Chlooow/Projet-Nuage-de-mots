// Fonction pour charger le .txt de motsvides
function chargerMotsVides() {
    //ou se trouve le fichier dans mon architecture
    $ficher = '../data/utils/motsvides.txt';

    if(!file_exists($fichier)) {
        return [];
    }

    $contenu = file_get_contents($fichier);
    $mots = explode("\n", $contenu);
    
    // Nettoyer et normaliser
    $mots = array_map('trim, $mots');
    $mots = array_map('mb_strtolover', $mots);
    $mots = array_filter($mots, function($mot) {
        return !empty($mot);
    });

    return array_values($mots);
}

// fonction pour nettoyer le texte
function nettoyerTexte($texte) {
    $texte = mb_strtolower($texte, 'UTF-8');

    // Supprimer les accents
    $texte = iconv('UTF-8', 'ASCII//TRANSLIT//IGNORE', $texte);
    if ($texte === false) {
        $texte = mb_strtolower($texte, 'UTF-8');
    }

    // Remplacer ponctuation et caractères spéciaux par des espaces
    $texte = preg_replace('/[^\p{L}\p{N}\s]/u', ' ', $texte);

    // Supprimer les chiffres
    $texte = preg_replace('/[0-9]+/', ' ', $texte);
    
    // Réduire les espaces multiples
    $texte = preg_replace('/\s+/', ' ', $texte);

    return trim($texte);
}

// tokenisation
function tokeniser($texte) {
    return explode(' ', $texte);
}

// filtrage des mots
function filtrerMots($mots, $motsvides, $longueurMin = 3) {
    return array_filter($mots, function($mot) use ($motsvides, $longueurMin) {
        return mb_strlen($mot) >= $longueurMin && !in_array($mot, $motsvides);
    });
}

function comptOcc($mots){
    $compteur = [];
    
    foreach ($mots as $mot) {
        if (!isset($compteur[$mot])) {
            $compteur[$mot] = 0;
        }
        $compteur[$mot]++;
    }
    
    // Trier par fréquence décroissante
    arsort($compteur);
    
    return $compteur;
}

// Traitement des requête AJAX
if($_SERVER['REQUEST_METHOD'] !== 'POST') {
    echo json_encode([
        'success' => false,
        'error' => 'Méthode non autorisée. Utilisez POST.'
    ]);
    exit;
}

// Récupérer le texte (JSON ou FormData)
$input = file_get_contents('php://input');
$data = json_decode($input, true);

$texteOriginal = '';

// Si JSON
if ($data && isset($data['texte'])) {
    $texteOriginal = $data['texte'];
}
// Si FormData
else if (isset($_POST['texte'])) {
    $texteOriginal = $_POST['texte'];
}

if (empty($texteOriginal)) {
    echo json_encode([
        'success' => false,
        'error' => 'Aucun texte fourni'
    ]);
    exit;
}

// traitement du texte
try {
    // 1. Charger les mots vides
    $motsvides = chargerMotsVides();
    
    // 2. Nettoyer le texte
    $texteNettoye = nettoyerTexte($texteOriginal);
    
    // 3. Tokeniser
    $mots = tokeniser($texteNettoye);
    
    // 4. Filtrer
    $motsFiltres = filtrerMots($mots, $motsvides);
    
    // 5. Compter
    $compteur = compterOccurrences($motsFiltres);
    
    // 6. Limiter aux 100 premiers
    $compteur = array_slice($compteur, 0, 100, true);
    
    // Statistiques
    $stats = [
        'totalMots' => count($mots),
        'motsUniques' => count(array_unique($mots)),
        'motsFiltres' => count($motsFiltres),
        'motsSignificatifs' => count($compteur),
        'motsVidesRetires' => count($mots) - count($motsFiltres),
        'nombreMotsVides' => count($motsvides)
    ];

    // Préparer les données pour wordcloud2.js (format: [[mot, freq], ...])
    $donneesNuage = [];
    foreach ($compteur as $mot => $freq) {
        $donneesNuage[] = [$mot, $freq];
    }
    
    // Validation finale
    if (count($donneesNuage) === 0) {
        echo json_encode([
            'success' => false,
            'error' => 'Aucun mot significatif trouvé dans le texte',
            'stats' => $stats
        ]);
        exit;
    }

    // Retourner le résultat
    echo json_encode([
        'success' => true,
        'donnees' => $donneesNuage,
        'statistiques' => $stats,
        'message' => 'Analyse terminée avec succès'
    ], JSON_UNESCAPED_UNICODE);
    
} catch (Exception $e) {
    echo json_encode([
        'success' => false,
        'error' => 'Erreur lors du traitement: ' . $e->getMessage()
    ]);

        btnGenerer.addEventListener('click', async () => {
            const texte = textarea.value;
            const validation = validerTexte(texte);
            if (!validation.valide) {
                afficherErreur(validation.message);
                return;
            }

            if (!verifierWordCloud()) {
                return;
            }
            console.log(`Envoi de ${validation.nombreMots} mots au serveur pour analyse...`);
            afficherChargement(true);

            try {
                const resultat = await analyserTexteServeur(texte);
                
                console.log('Réponse du serveur reçue');
                
                if (!resultat.donnees || resultat.donnees.length === 0) {
                    afficherErreur('Aucun mot significatif trouvé dans le texte');

                    return;

                afficherStatistiques(resultat.statistiques);
                
                genererNuageVisuel(resultat.donnees);
                
                afficherSucces('Nuage de mots généré avec succès !');

                console.error('Erreur lors de la génération:', error);
                afficherErreur('Erreur lors de la génération du nuage: ' + error.message);
                nettoyerResultat();
            }

    // generer le visuelle du nuage
    function genererNuageVisuel(donneesNuage) {
        if (!boiteResultat) return;
        
        // Créer le canvas
        boiteResultat.innerHTML = '<canvas id="nuage-canvas"></canvas>';
        const canvas = document.getElementById('nuage-canvas');
        
        if (!canvas) {
            afficherErreur('Impossible de créer le canvas');
            return;
        }
        
        // Ajuster la taille du canvas à son conteneur
        const largeur = boiteResultat.offsetWidth - 60;
        const hauteur = boiteResultat.offsetHeight - 60;
        
        canvas.width = largeur > 0 ? largeur : 600;
        canvas.height = hauteur > 0 ? hauteur : 400;
        
        console.log(`Generation du nuage: ${canvas.width}x${canvas.height}px`);
        
        // Configuration de wordcloud2.js
        try {
            WordCloud(canvas, {
                list: donneesNuage,
                gridSize: 8,
                weightFactor: function(size) {
                    return Math.pow(size, 0.7) * canvas.width / 40;
                },
                fontFamily: 'Comic Sans MS, Arial, sans-serif',
                color: function() {
                    // Palette de couleurs bleues
                    const couleurs = [
                        '#00BFFF',  // DeepSkyBlue
                        '#1E90FF',  // DodgerBlue
                        '#4169E1',  // RoyalBlue
                        '#0099CC',  // Blue Ciel
                        '#5F9EA0',  // CadetBlue
                        '#4682B4',  // SteelBlue
                        '#87CEEB',  // SkyBlue
                        '#4682B4'   // SteelBlue
                    ];
                    return couleurs[Math.floor(Math.random() * couleurs.length)];
                },
                rotateRatio: 0.3,
                rotationSteps: 2,
                backgroundColor: 'white',
                minSize: 12,
                drawOutOfBound: false,
                shrinkToFit: true,
                hover: function(item) {
                    if (item) {
                        canvas.style.cursor = 'pointer';
                    } else {
                        canvas.style.cursor = 'default';
                    }
                },
                click: function(item) {
                    if (item) {
                        console.log(`Mot cliqué: "${item[0]}" (${item[1]} occurrences)`);
                    }
                }
            });
            
            console.log('Nuage de mots affiché');
            } catch (error) {
            console.error(' Erreur WordCloud:', error);
            afficherErreur('Erreur lors de l\'affichage du nuage');
        }
